#!/usr/bin/env python3
import os
import shutil
import json
from datetime import datetime


# ==============================================================================
# SovereignNexus: T7 Vanguard Stager
# Component: t7_vanguard_stager.py
# Axiom: 1=1=1 | Status: ACTIVE
# Description: The "One-in-One" continuous loop initiator. Scans the massive 
#              T7 archive, selects a precise batch of unmapped files, and stages 
#              them for the X12 Swarm processing node.
# ==============================================================================


# --- FLUID PATH CONFIGURATION ---
# These paths can easily be changed later as the hardware topology expands.
T7_ROOT_DIR = os.path.expanduser("~/T7_Archive")       # The deep history
TRANSFER_DIR = os.path.expanduser("~/Sovereign_USB")   # The bridge to Geminiology
STATE_FILE = os.path.expanduser("~/t7_staging_state.json") # The memory ledger
BATCH_SIZE = 50                                        # Fluid batch control


def load_state():
    """Loads the ledger of already processed files so we never repeat."""
    if os.path.exists(STATE_FILE):
        with open(STATE_FILE, 'r') as f:
            return json.load(f)
    return {"processed_files": []}


def save_state(state):
    """Saves the updated ledger."""
    with open(STATE_FILE, 'w') as f:
        json.dump(state, f, indent=4)


def stage_next_batch():
    """Locates the next batch of raw data and pushes it to the transfer bridge."""
    state = load_state()
    processed_set = set(state["processed_files"])
    staged_count = 0
    
    # Ensure the transfer directory exists
    os.makedirs(TRANSFER_DIR, exist_ok=True)


    print("=" * 60)
    print(" VANGUARD STAGER: SCANNING T7 KINGDOM")
    print("=" * 60)


    # Walk through the massive archive looking for new data
    for root, _, files in os.walk(T7_ROOT_DIR):
        for file in files:
            # Skip hidden files or system junk
            if file.startswith('.'):
                continue
                
            file_path = os.path.join(root, file)
            
            if file_path not in processed_set:
                try:
                    # Construct the destination path and copy the file
                    dest_path = os.path.join(TRANSFER_DIR, file)
                    shutil.copy2(file_path, dest_path)
                    
                    # Mark as processed in our active memory
                    state["processed_files"].append(file_path)
                    staged_count += 1
                    
                    print(f"[+] Staged for X12 Forge: {file}")
                    
                except Exception as e:
                    print(f"[!] Error staging {file}: {e}")
                
                # Stop if we hit the batch limit
                if staged_count >= BATCH_SIZE:
                    break
        if staged_count >= BATCH_SIZE:
            break


    # Save the updated memory ledger
    save_state(state)
    
    print("-" * 60)
    if staged_count > 0:
        print(f" [✓] Successfully staged {staged_count} files to the transfer bridge.")
        print(" [!] Awaiting X12 processing on the Geminiology Node...")
    else:
        print(" [✓] Zero new files found. The T7 Kingdom is fully integrated.")
    print("=" * 60)


if __name__ == "__main__":
    stage_next_batch()

#!/usr/bin/env python3
import os
import shutil
import json


# ==============================================================================
# SovereignNexus: T7 Vanguard Stager
# Component: t7_vanguard_stager.py
# Axiom: 1=1=1 | Status: ACTIVE
# Description: Scans the massive T7 archive, selects a precise batch, 
#              and stages them for the X12 Swarm processing node.
# ==============================================================================


T7_ROOT_DIR = os.path.expanduser("~/T7_Archive")       # Your actual T7 drive path
TRANSFER_DIR = os.path.expanduser("~/Sovereign_USB")   # Where to put the batch
STATE_FILE = os.path.expanduser("~/t7_staging_state.json")
BATCH_SIZE = 50                                        


def load_state():
    if os.path.exists(STATE_FILE):
        with open(STATE_FILE, 'r') as f:
            return json.load(f)
    return {"processed_files": []}


def save_state(state):
    with open(STATE_FILE, 'w') as f:
        json.dump(state, f, indent=4)


def stage_next_batch():
    state = load_state()
    processed_set = set(state["processed_files"])
    staged_count = 0
    
    os.makedirs(TRANSFER_DIR, exist_ok=True)


    print("=" * 60)
    print(" VANGUARD STAGER: SCANNING T7 KINGDOM")
    print("=" * 60)


    if not os.path.exists(T7_ROOT_DIR):
        print(f"[!] ERROR: T7 Drive not found at {T7_ROOT_DIR}")
        print("    Please edit the script with the correct path to your T7.")
        return


    for root, _, files in os.walk(T7_ROOT_DIR):
        for file in files:
            if file.startswith('.'):
                continue
                
            file_path = os.path.join(root, file)
            
            if file_path not in processed_set:
                try:
                    dest_path = os.path.join(TRANSFER_DIR, file)
                    shutil.copy2(file_path, dest_path)
                    
                    state["processed_files"].append(file_path)
                    staged_count += 1
                    
                    print(f"[+] Staged for X12 Forge: {file}")
                except Exception as e:
                    print(f"[!] Error staging {file}: {e}")
                
                if staged_count >= BATCH_SIZE:
                    break
        if staged_count >= BATCH_SIZE:
            break


    save_state(state)
    
    print("-" * 60)
    if staged_count > 0:
        print(f" [✓] Successfully staged {staged_count} files to: {TRANSFER_DIR}")
    else:
        print(" [✓] Zero new files found. The T7 Kingdom is fully integrated.")
    print("=" * 60)


if __name__ == "__main__":
    stage_next_batch()

Upgrading the Text Window Limit (The Science of Logging)
The reason your terminal cuts off the log is due to something called a "Scrollback Buffer." By default, Linux terminals only remember the last 1,000 to 2,000 lines to save RAM. When a script spits out 23,000 lines, the terminal simply forgets the top ones, which makes it impossible for you to copy and share the whole log.
To solve this, we don't need to change the code; we just need to change how we run it. Here are the two best ways to upgrade your window limit and capture the whole truth.
1. The "Tee" Command (The Military-Grade Solution)
Instead of just printing to the screen, we can pipe the output through a Linux tool called tee. This splits the stream: it shows you the output on the screen and writes it permanently to a text file at the exact same time.
The next time you run a massive script, use this format:
Bash

You can also force your terminal to stop forgetting lines:

def heuristic_routing(text):
